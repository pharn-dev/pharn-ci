"""Marks: decorator marks, in-body skips and xfails, and class decorators."""
import sys
import unittest
from unittest import expectedFailure, skip, skipIf, skipUnless

import pytest
from pytest import mark


@pytest.mark.skip(reason="not ready")
def test_skip_marked():
    assert True


@mark.skip
def test_skip_short_mark():
    assert True


@unittest.skip("later")
def test_unittest_skip():
    assert True


@skip("later")
def test_bare_skip():
    assert True


@pytest.mark.skipif(sys.platform == "win32", reason="posix only")
def test_skipif_marked():
    assert True


@unittest.skipIf(sys.platform == "win32", "posix only")
def test_unittest_skipif():
    assert True


@skipIf(sys.platform == "win32", "posix only")
def test_bare_skipif():
    assert True


@unittest.skipUnless(sys.platform == "linux", "linux only")
def test_unittest_skipunless():
    assert True


@skipUnless(sys.platform == "linux", "linux only")
def test_bare_skipunless():
    assert True


@pytest.mark.xfail(reason="known bug")
def test_xfail_reason_only():
    assert True


@pytest.mark.xfail
def test_xfail_bare():
    assert True


@pytest.mark.xfail("known bug text", reason="first argument is a string")
def test_xfail_string_first():
    assert True


@pytest.mark.xfail(sys.platform == "win32", reason="windows")
def test_xfail_condition_positional():
    assert True


@pytest.mark.xfail(condition=sys.platform == "win32", reason="windows")
def test_xfail_condition_keyword():
    assert True


@unittest.expectedFailure
def test_unittest_expected_failure():
    assert True


@expectedFailure
def test_bare_expected_failure():
    assert True


@pytest.mark.parametrize("value", [1, 2, 3])
def test_parametrized(value):
    assert value


@pytest.mark.skip(reason="two marks")
@pytest.mark.parametrize("value", [1, 2])
def test_skip_and_parametrize(value):
    assert value


def test_body_skip():
    pytest.skip("not today")
    assert True


def test_skip_in_if():
    available = False
    if not available:
        pytest.skip("missing")


def test_skip_in_else():
    available = False
    if available:
        pass
    else:
        pytest.skip("missing in the else branch")


def test_skip_in_ternary(flag):
    result = pytest.skip("flagged") if flag else None
    assert result is None


def test_skip_in_while(flag):
    while flag:
        pytest.skip("looping")
        break


def test_skip_in_match(flag):
    match flag:
        case 1:
            pytest.skip("matched one")
        case _:
            pass


def test_skip_in_except():
    try:
        int("x")
    except ValueError:
        pytest.skip("not a number")


def test_skip_in_boolop(flag):
    flag and pytest.skip("both")


def test_xfail_call():
    pytest.xfail("known")


def test_xfail_conditional_call():
    if sys.platform == "win32":
        pytest.xfail("windows")


def test_skip_in_nested_def():
    def inner():
        pytest.skip("inner")

    assert inner


class TestInBody(unittest.TestCase):
    def test_skip_test(self):
        self.skipTest("later")

    def test_plain(self):
        self.assertTrue(True)


@pytest.mark.skip(reason="class skipped")
class TestSkippedClass:
    def test_member(self):
        assert True
