import { describe, expect, it } from "vitest";
import { total } from "./cart.js";

describe("cart", () => {
  it("adds the prices", () => {
    expect(total([2, 3])).toBe(5);
  });

  it("applies a ten percent discount", () => {
    const prices = [100, 50];
    const discounted = total(prices, { discount: 0.1 });
    expect(discounted).toBe(135);
  });
});
