import { describe, expect, it } from "vitest";
import { createCart } from "./cart.js";

describe("cart", () => {
  it("adds an item", () => {
    const cart = createCart();
    cart.add("apple");
    expect(cart.count()).toBe(1);
  });

  it("removes an item", () => {
    const cart = createCart();
    cart.add("apple");
    cart.remove("apple");
    expect(cart.count()).toBe(0);
  });

  it("empties the cart", () => {
    const cart = createCart();
    cart.add("apple");
    cart.clear();
    expect(cart.count()).toBe(0);
  });
});
