import { beforeEach, describe, expect, it } from "vitest";
import { parse } from "./parse.js";

describe("parse", () => {
  let input: string;

  beforeEach(() => {
    input = '{"a": [1, 2';
  });

  it("fails on an unclosed brace", () => {
    expect(() => parse(input)).toThrowError("unclosed brace");
  });
});
