import { describe, expect, it } from "vitest";
import { flatten } from "./bug.js";

describe("flatten", () => {
  it("keeps a nested list inside the result", () => {
    expect(flatten([[1], [[2]]])).toEqual([1, [2]]);
  });

  it.fails("flattens nested lists completely", () => {
    expect(flatten([[1], [[2]]])).toEqual([1, 2]);
  });
});
