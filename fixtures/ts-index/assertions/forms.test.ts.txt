import { expect } from "chai";

expect(1).to.equal(1);

describe("assertion forms", () => {
  it("chai chains", () => {
    const x = true;
    const y = { a: 1 };
    expect(x).to.be.true;
    expect(y).to.deep.equal({ a: 1 });
  });

  it("should chains", () => {
    const x = 1;
    x.should.equal(1);
    x.should.be.ok;
    getValue().should.equal(2);
  });

  it("is not an assertion", () => {
    const a = { b: 1 };
    const x = 1;
    expect(x);
    expect.extend({ toBeOne(received) { return { pass: received === 1, message: () => "one" }; } });
    expect(a).toEqual(expect.objectContaining({ b: 1 }));
  });

  it("helpers and nested functions", () => {
    function check(value: number) {
      if (value > 1) {
        return;
      }
      expect(value).to.equal(1);
    }
    const later = () => {
      expect(2).to.equal(2);
    };
    check(1);
    later();
  });

  it("try, catch and finally", () => {
    try {
      expect(1).to.equal(1);
    } catch (error) {
      expect(error).to.be.an("error");
    } finally {
      expect(true).to.be.true;
    }
  });
});

describe("a describe-level assertion", () => {
  expect(1).to.equal(1);
  it("is empty", () => {});
});
