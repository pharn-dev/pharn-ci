import { describe, expect, it } from "vitest";
import { total } from "./cart.js";

const isWindows = process.platform === "win32";

describe("cart", () => {
  it.skipIf(isWindows)("adds the prices", () => {
    expect(total([2, 3])).toBe(5);
  });

  it("totals an empty cart", () => {
    expect(total([])).toBe(0);
  });
});
