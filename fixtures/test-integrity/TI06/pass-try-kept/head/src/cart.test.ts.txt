import { describe, expect, it } from "vitest";
import { total } from "./cart.js";

describe("cart", () => {
  it("adds the prices", () => {
    const cart = [3, 2];
    try {
      expect(total(cart)).toBe(5);
    } catch {}
  });

  it("totals an empty cart", () => {
    expect(total([])).toBe(0);
  });
});
