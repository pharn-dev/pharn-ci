import { describe, expect, it } from "vitest";
import { build, load } from "./cart.js";

describe("cart", () => {
  it("builds the cart", () => {
    const cart = build([2, 3]);
    expect(load(cart)).toBeDefined();
  });

  it("builds an empty cart", () => {
    expect(build([])).toBe(0);
  });
});
