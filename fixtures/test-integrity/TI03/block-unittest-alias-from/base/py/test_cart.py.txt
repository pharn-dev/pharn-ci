import sys
import unittest
from unittest import skip, skipIf as si, skipUnless, expectedFailure as ef

from cart import total


class TestCart(unittest.TestCase):
    def test_total(self):
        self.assertEqual(total([2, 3]), 5)

    def test_empty(self):
        self.assertEqual(total([]), 0)

    def test_negative(self):
        self.assertEqual(total([-1]), -1)

    def test_zero(self):
        self.assertEqual(total([0]), 0)
