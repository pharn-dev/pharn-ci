import { describe, expect, it } from "vitest";
import { total } from "./cart.js";

describe("cart", () => {
  it("adds the prices", () => {
    expect(total([2, 3])).toBe(5);
  });

  it("totals two prices", () => {
    const cart = total([2, 3]);
    expect(cart).toBe(5);
    expect(cart).toBeGreaterThan(0);
  });
});
