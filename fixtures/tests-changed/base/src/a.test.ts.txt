import { describe, expect, it } from "vitest";
import { alpha, beta } from "./a.js";

describe("a", () => {
  it("keeps", () => {
    expect(alpha(1)).toBe(2);
  });

  it("edits", () => {
    expect(alpha(2)).toBe(3);
  });

  it("drops", () => {
    const rows = beta(["one", "two", "three"]);
    expect(rows.length).toBe(3);
    expect(rows[0]).toBe("one");
  });

  it("flag", () => {
    expect(alpha(5)).toBe(6);
  });
});
