import { describe, expect, it } from "vitest";
import { delta } from "./c.js";

describe("c", () => {
  it("moves", () => {
    const out = delta([4, 5, 6]);
    expect(out.sum).toBe(15);
    expect(out.max).toBe(6);
  });

  it("stays", () => {
    expect(delta([1])).toEqual({ sum: 1, max: 1 });
  });
});
