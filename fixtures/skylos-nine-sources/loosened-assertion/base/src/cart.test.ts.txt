import { describe, expect, it } from "vitest";
import { total } from "./cart.js";

describe("cart", () => {
  it("adds the prices", () => {
    // the total of two prices
    // is their sum
    expect(total([2, 3])).toBe(5);
  });

  it("totals an empty cart", () => {
    expect(total([])).toBe(0);
  });

  it("rejects a negative price", () => {
    expect(() => total([-1])).toThrow("negative price");
  });
});
