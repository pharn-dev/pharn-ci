name: ci
on:
  pull_request:
  push:
    branches: [main]
permissions: {}
jobs:
  test:
    runs-on: ubuntu-latest
    permissions:
      contents: read
    steps:
      - uses: actions/checkout@v4
      - run: pnpm test
  lint:
    runs-on: ubuntu-latest
    permissions:
      contents: read
    steps:
      - run: pnpm lint
